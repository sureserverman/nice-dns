[Unit]
Description=nice-dns DNS stack pod
After=nice-dns-network.service podman-user-wait-network-online.service

[Pod]
PodName=nice-dns
# Network=nice-dns.network is load-bearing even though pod members talk over
# localhost: the network unit injects DNS=1.1.1.1 into containers' resolv.conf,
# which is required so pod members can resolve image registries / Tor bridges
# at startup. The host resolver is pinned to 127.0.0.1 (the pod itself), so
# without this the pod would deadlock its own bootstrap.
Network=nice-dns.network
PublishPort=53:53/tcp
PublishPort=53:53/udp
PublishPort=8880:80
# ExitPolicy is not a recognized [Pod] key in Podman quadlet (5.4.x rejects it
# with "unsupported key 'ExitPolicy' in group 'Pod'", which silently drops the
# whole nice-dns-pod.service generation). Pass it through PodmanArgs instead so
# `podman pod create --exit-policy=continue` is what actually runs.
PodmanArgs=--exit-policy=continue

[Service]
RestartSec=3s
TimeoutStopSec=70

[Install]
WantedBy=default.target
